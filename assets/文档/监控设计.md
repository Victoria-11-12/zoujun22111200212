    name: 'http_requests_total',
    help: 'Total HTTP requests',
    labelNames: ['service', 'method', 'endpoint', 'status'],
    registers: [register]

const httpRequestDuration = new client.Histogram({
    name: 'http_request_duration_seconds',
    help: 'HTTP request duration in seconds',
    labelNames: ['service', 'endpoint'],
    buckets: [0.01, 0.05, 0.1, 0.5, 1, 2, 5],
    registers: [register]
});

// 登录注册计数器
const authCounter = new client.Counter({
    name: 'auth_requests_total',
    help: '登录注册请求总数',
    labelNames: ['type', 'status'],
    registers: [register]
});

// 请求统计中间件
app.use((req, res, next) => {
    const start = Date.now();
    res.on('finish', () => {
        const duration = (Date.now() - start) / 1000;
        const endpoint = req.route ? req.route.path : req.path;
        httpRequestsTotal.inc({
            service: 'nodejs',
            method: req.method,
            endpoint: endpoint,
            status: res.statusCode
        });
        httpRequestDuration.observe({
            service: 'nodejs',
            endpoint: endpoint
        }, duration);
    });
    next();
});

// 登录接口埋点
if (err) {
    authCounter.inc({ type: 'login', status: 'fail' });
    return res.send({ code: 500, msg: '服务器数据库错误' });
}
if (results.length === 0) {
    authCounter.inc({ type: 'login', status: 'fail' });
    return res.send({ code: 400, msg: '该用户不存在' });
}
if (!isMatch) {
    authCounter.inc({ type: 'login', status: 'fail' });
    return res.send({ code: 400, msg: '密码错误' });
}
authCounter.inc({ type: 'login', status: 'success' });

// 注册接口埋点
if (err) {
    authCounter.inc({ type: 'register', status: 'fail' });
    return res.send({ code: 500, msg: '注册失败' });
}
authCounter.inc({ type: 'register', status: 'success' });

## 七、FastAPI Agent Token 监控

### 0. 背景说明

系统有四个完整的 Agent 流程，需要统计每个 Agent 消耗的 LLM token 数量（input/output），通过 Prometheus 指标暴露，可在 Grafana 中查看。

四个 Agent：
1. **用户 Agent** — intent_chain + direct/warning/sql+wrap
2. **Admin Agent** — admin_intent_chain + admin_warning/admin_executor
3. **Chart Agent** — chart_intent_chain + chart_not/chart_workflow
4. **Eval Agent** — response_eval_chain + code_eval_chain

核心思路：使用 LangChain 的 AsyncCallbackHandler + stream_usage=True，在每次 LLM 调用结束时自动捕获 token 用量，按 agent 名称累加到 Prometheus 指标。

### 1. 新建 app/metrics.py
```python
from prometheus_client import Counter, Histogram, CollectorRegistry

registry = CollectorRegistry()

http_requests_total = Counter('http_requests_total', 'Total HTTP requests',
    ['service', 'method', 'endpoint', 'status'], registry=registry)

http_request_duration_seconds = Histogram('http_request_duration_seconds',
    'HTTP request duration in seconds', ['service', 'endpoint'],
    buckets=[0.01, 0.05, 0.1, 0.5, 1, 2, 5], registry=registry)

# Agent token 消耗指标
agent_token_usage_total = Counter('agent_token_usage_total',
    'Total token usage by agent', ['agent', 'token_type'], registry=registry)
```

### 2. 新建 app/token_tracker.py
```python
from langchain_core.callbacks import AsyncCallbackHandler
from app.metrics import agent_token_usage_total

class TokenTrackerCallback(AsyncCallbackHandler):
    def __init__(self, agent_name: str):
        self.agent_name = agent_name

    async def on_llm_end(self, response, **kwargs):
        usage = {}
        if response.llm_output:
            usage = response.llm_output.get("usage", {})
        if not usage:
            for gen_list in response.generations:
                for gen in gen_list:
                    msg = getattr(gen, 'message', None)
                    if msg and hasattr(msg, 'usage_metadata') and msg.usage_metadata:
                        usage = msg.usage_metadata
                        break
        input_tokens = usage.get("input_tokens", 0) or usage.get("prompt_tokens", 0)
        output_tokens = usage.get("output_tokens", 0) or usage.get("completion_tokens", 0)
        if input_tokens:
            agent_token_usage_total.labels(agent=self.agent_name, token_type="input").inc(input_tokens)
        if output_tokens:
            agent_token_usage_total.labels(agent=self.agent_name, token_type="output").inc(output_tokens)
```

## 九、遇到的问题及解决

### 1. Grafana 启动闪退
**原因**: 缺少 homepath 参数
**解决**: 添加 `-homepath .` 参数启动
```
grafana.exe server -config .\conf\defaults.ini -homepath .
```

### 2. Flask /metrics 404
**原因**: prometheus_flask_exporter 在 debug=True 模式下子进程重载导致端点丢失
**解决**: 改用 prometheus_client 手动实现指标和 /metrics 端点

### 3. Grafana 查询语法错误
**错误**: `bad_data: invalid parameter "query": 1:28: parse error`
**原因**: 标签值没有用引号包裹
**解决**: 使用正确语法 `http_requests_total{service="nodejs"}`

### 4. Node.js prom-client 模块未找到
**解决**: 在 Web_Node 目录安装依赖 `npm install prom-client`

### 5. 管理员界面 404
**原因**: 直接访问 localhost:3000 没有走 Nginx 代理
**解决**: 通过 Nginx 访问 http://localhost/admin.html
